┌─────────────────────────────────────────────────────────────────────────────┐
│ 阶段1：初始化 - hwcontext_rkmpp.c 负责                                       │
├─────────────────────────────────────────────────────────────────────────────┤
│ 创建内存池（MppBufferGroup），类型为 MPP_BUFFER_TYPE_DRM（DMA-BUF）           │
│ 解码器后续从这个池子里拿内存来写解码数据                                        │
└─────────────────────────────────────────────────────────────────────────────┘
                                    │
                                    ▼
┌─────────────────────────────────────────────────────────────────────────────┐
│ 阶段2：解码输入 - rkmppdec.c 的 rkmpp_send_packet()                           │
├─────────────────────────────────────────────────────────────────────────────┤
│ AVPacket → 转换成 MppPacket → mpi->decode_put_packet() 送入硬件               │
└─────────────────────────────────────────────────────────────────────────────┘
                                    │
                                    ▼
┌─────────────────────────────────────────────────────────────────────────────┐
│ 阶段3：硬件解码                                                              │
├─────────────────────────────────────────────────────────────────────────────┤
│ MPP 硬件从内存池拿一块 DMA-BUF，把解码后的 YUV 数据写进去                        │
│ 返回 MppFrame，里面包含：                                                     │
│   - MppBuffer（指向那块 DMA-BUF）                                            │
│   - 宽高、stride、格式等信息                                                  │
└─────────────────────────────────────────────────────────────────────────────┘
                                    │
                                    ▼
┌─────────────────────────────────────────────────────────────────────────────┐
│ 阶段4：封装输出 - rkmppdec.c 的 rkmpp_export_frame()                          │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  MppFrame 里面有：                                                           │
│  ┌─────────────────────────────────────────────────────────────────────┐    │
│  │ MppBuffer mpp_buf = mpp_frame_get_buffer(mpp_frame);                 │    │
│  │ int fd = mpp_buffer_get_fd(mpp_buf);                                 │    │
│  │ int width = mpp_frame_get_width(mpp_frame);                          │    │
│  │ int height = mpp_frame_get_height(mpp_frame);                        │    │
│  │ int stride = mpp_frame_get_hor_stride(mpp_frame);                    │    │
│  └─────────────────────────────────────────────────────────────────────┘    │
│                                    │                                        │
│                                    ▼                                        │
│  填充到 AVRKMPPDRMFrameDescriptor：                                         │
│  ┌─────────────────────────────────────────────────────────────────────┐    │
│  │ typedef struct AVRKMPPDRMFrameDescriptor {                           │    │
│  │     AVDRMFrameDescriptor drm_desc;  // 里面有 objects[0].fd          │    │
│  │     MppBuffer buffers[4];            // 保存 MppBuffer 引用          │    │
│  │ } AVRKMPPDRMFrameDescriptor;                                        │    │
│  │                                                                     │    │
│  │ desc->buffers[0] = mpp_buf;                                         │    │
│  │ desc->drm_desc.objects[0].fd = fd;                                  │    │
│  │ desc->drm_desc.layers[0].planes[0].pitch = stride;                  │    │
│  └─────────────────────────────────────────────────────────────────────┘    │
│                                    │                                        │
│                                    ▼                                        │
│  挂到 AVFrame：                                                             │
│  ┌─────────────────────────────────────────────────────────────────────┐    │
│  │ frame->data[0] = (uint8_t *)desc;  // 指向描述符，不是像素数据！      │    │
│  │ frame->format = AV_PIX_FMT_DRM_PRIME;                               │    │
│  └─────────────────────────────────────────────────────────────────────┘    │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘
                                    │
                                    ▼
┌─────────────────────────────────────────────────────────────────────────────┐
│ 阶段5：用户获取                                                              │
├─────────────────────────────────────────────────────────────────────────────┤
│ avcodec_receive_frame() 返回 AVFrame                                        │
│                                                                             │
│ 用户代码：                                                                   │
│ AVRKMPPDRMFrameDescriptor *desc = (void *)frame->data[0];                   │
│ int dma_fd = desc->drm_desc.objects[0].fd;  // 拿到 fd！                    │
│                                                                             │
│ // 接下来可以直接给 RGA/DRM 用，零拷贝                                        │
└─────────────────────────────────────────────────────────────────────────────┘
